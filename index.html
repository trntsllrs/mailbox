        const memGrid = document.getElementById('mem-grid');

        const memIcons = ['🎁','🎁','🎁','💌','💌','💌','📄','📄','📄'];

        memIcons.sort(() => Math.random() - 0.5);

        

        let flippedCards = [];

        memIcons.forEach((icon, index) => {

            const card = document.createElement('div');

            card.className = 'mem-card';

            card.dataset.icon = icon;

        function flipMemCard(card) {

            if (card.classList.contains('flipped') || card.classList.contains('matched') || flippedCards.length >= 3) return;

            

            card.classList.add('flipped');

            card.innerText = card.dataset.icon;

            flippedCards.push(card);


            if (flippedCards.length === 3) {

                const first = flippedCards[0].dataset.icon;

                const isMatch = flippedCards.every(c => c.dataset.icon === first);


                if (isMatch) {

                    flippedCards.forEach(c => c.classList.add('matched'));

                    flippedCards = [];

                    addScore(1);

                } else {

                    setTimeout(() => {

                        flippedCards.forEach(c => {

                            c.classList.remove('flipped');

                            c.innerText = '📮';

                        });

                        flippedCards = [];

                    }, 700);

                }

            }

        }

        const shooStage = document.getElementById('shoo-stage');

        let shooCount = 0;

        let shooTimeLeft = 10;

        let shooTimer = null;

        let shooActive = true;


        function startShooGame() {

            if (shooTimer) return;

            shooTimer = setInterval(() => {

                shooTimeLeft--;

                updateShooStatus();

                if (shooTimeLeft <= 0) {

                    clearInterval(shooTimer);

                    shooActive = false;

                    document.querySelectorAll('.shoo-visitor').forEach(v => v.remove());

                    addScore(shooCount); // 1 point per shooed visitor

                    document.getElementById('s3-status').innerText = `✓ Blitz over! You shooed ${shooCount} visitors! (+${shooCount})`;

                    document.getElementById('s3-status').style.color = 'var(--success-green)';

                }

            }, 1000);

        }


        function updateShooStatus() {

            if (shooActive) {

                document.getElementById('s3-status').innerText = `⏱️ ${shooTimeLeft}s left | Shooed: ${shooCount}`;

            }

        }

            const y = 75 + Math.sin(rad) * 55 - 15;

            

            visitor.style.left = `${x}px`;

            visitor.style.top = `${y}px`;


            visitor.onclick = () => {

        /* --- STANZA 4: Moving Rain Grid Maze --- */

        const gridBoard = document.getElementById('grid-board');

        let playerPos = { x: 0, y: 0 };

        const goalPos = { x: 4, y: 4 };

        let rainDrops = [

            { x: 1, y: 0, dir: 1 },

            { x: 2, y: 4, dir: -1 },

            { x: 3, y: 1, dir: 1 },

            { x: 0, y: 2, dir: 1 }

        ];

        let mazeActive = true;


        for (let r = 0; r < 5; r++) {

            for (let c = 0; c < 5; c++) {

                const cell = document.createElement('div');

                cell.className = 'grid-cell';

                cell.id = `cell-${c}-${r}`;

                if (c === goalPos.x && r === goalPos.y) cell.classList.add('goal');

                gridBoard.appendChild(cell);

            }

        }


        function renderMaze() {

            for (let r = 0; r < 5; r++) {

                for (let c = 0; c < 5; c++) {

                    const cell = document.getElementById(`cell-${c}-${r}`);

                    cell.innerText = (c === goalPos.x && r === goalPos.y) ? '📬' : '';

                }

            }

            rainDrops.forEach(drop => {

                const cell = document.getElementById(`cell-${drop.x}-${drop.y}`);

                if (cell && (drop.x !== goalPos.x || drop.y !== goalPos.y)) cell.innerText = '💧';

            });

            const pCell = document.getElementById(`cell-${playerPos.x}-${playerPos.y}`);

            if (pCell) pCell.innerText = '✉️';

        }


        function movePlayer(dx, dy) {

            if (!mazeActive) return;

            const newX = Math.max(0, Math.min(4, playerPos.x + dx));

            const newY = Math.max(0, Math.min(4, playerPos.y + dy));

            playerPos = { x: newX, y: newY };

            checkMazeState();

            renderMaze();

        }


        function updateRain() {

            if (!mazeActive) return;

            rainDrops.forEach(drop => {

                drop.y += drop.dir;

                if (drop.y > 4) { drop.y = 4; drop.dir = -1; }

                if (drop.y < 0) { drop.y = 0; drop.dir = 1; }

            });

            checkMazeState();

            renderMaze();

        }


        function checkMazeState() {

            const hitRain = rainDrops.some(drop => drop.x === playerPos.x && drop.y === playerPos.y);

            if (hitRain) playerPos = { x: 0, y: 0 };

            

            if (playerPos.x === goalPos.x && playerPos.y === goalPos.y) {

                mazeActive = false;

                document.getElementById('s4-status').innerText = '✓ Safe through the storm! (+2)';

                document.getElementById('s4-status').style.color = 'var(--success-green)';

                addScore(2);

            }

        }


        setInterval(updateRain, 650);

        renderMaze();


        window.addEventListener('keydown', (e) => {

            if (['ArrowUp', 'KeyW'].includes(e.code)) movePlayer(0, -1);

            if (['ArrowDown', 'KeyS'].includes(e.code)) movePlayer(0, 1);

            if (['ArrowLeft', 'KeyA'].includes(e.code)) movePlayer(-1, 0);

            if (['ArrowRight', 'KeyD'].includes(e.code)) movePlayer(1, 0);

        });

        const secretWord = wordPool[Math.floor(Math.random() * wordPool.length)];

        let guessedLetters = [];

        let wrongGuesses = 0;

        const maxWrong = 5;

        const wordSlots = document.getElementById('word-slots');

            letterGrid.appendChild(btn);

        function guessHangman(char, btn) {

            btn.disabled = true;

            if (secretWord.includes(char)) {

                guessedLetters.push(char);

                updateSlots();

                if (secretWord.split('').every(c => guessedLetters.includes(c))) {

                    addScore(2);

                    disableHangman('SOLVED! (+2)');

                }

            } else {

                wrongGuesses++;

                if (wrongGuesses <= maxWrong) {

                    document.getElementById(`env-${wrongGuesses}`).classList.add('drawn');

                }

                if (wrongGuesses === maxWrong) {

                    disableHangman(`SEALED! WORD WAS: ${secretWord}`);

                }

            }

        }


        function updateSlots() {

            wordSlots.innerText = secretWord.split('')

                .map(c => guessedLetters.includes(c) ? c : '_')

                .join(' ');

        }


        function disableHangman(msg) {

            document.querySelectorAll('.key-btn').forEach(b => b.disabled = true);

            const sub = document.createElement('p');

            sub.className = 'typewriter-sub';

            sub.style.fontWeight = 'bold';

            sub.style.color = 'var(--postal-blue)';

            sub.innerText = msg;

            document.querySelector('.hangman-area').appendChild(sub);

        }
